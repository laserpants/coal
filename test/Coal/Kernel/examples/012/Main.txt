Main.op_id(dict : record({ id : * | op : */*/* | * }), a : *) =
  match<*>(dict : record({ id : * | op : */*/* | * })) {
    | ( $Record : { id : * | op : */*/* | * }/record({ id : * | op : */*/* | * })
      , r : { id : * | op : */*/* | * }
      ) =>
        select
          { id = group_id : * | q : { op : */*/* | * } } =
            r : { id : * | op : */*/* | * }
          in
            select
              { op = group_op : */*/* | _ : * } =
                q : { op : */*/* | * }
              in
                @<*>
                  ( group_op : */*/*
                  , group_id : *
                  , a : *
                  )
  }

          | {}
          }

      let
        r : int32 =
          @<int32>
            ( Main.op_id : record({ id : int32 | op : int32/int32/int32 | * })/int32/int32
            , d : record({ id : int32 | op : int32/int32/int32 | inv : int32/int32 | {} })

          (fn(a : *) => 0)
